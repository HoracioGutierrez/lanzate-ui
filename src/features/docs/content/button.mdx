## Usage

```tsx
import { Button } from "@/components/ui/button"

// Default
<Button>Take off</Button>

// Variant and color
<Button variant="outline" color="success">
  Confirmed
</Button>

// Icon and loading state
<Button isLoading={isPending} loadingText="Saving..." startIcon={<Save />}>
  Save
</Button>

// Elevated, with a tooltip
<Button effect="elevated" tooltip="Main action">
  Publish
</Button>

// As a link
<Button render={<a href="/docs" />}>Read the docs</Button>
```
